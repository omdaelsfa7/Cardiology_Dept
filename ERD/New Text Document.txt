Table Employees {
  EmployeeID int [pk]
  FullName varchar
  JobTitle varchar
  Salary decimal
  PhoneNumber varchar
  HomeAddress text
  JoiningDate date
}

Table Departments {
  DepartmentID int [pk]
  DepartmentCode varchar [unique]
  DepartmentName varchar [unique]
  ChairmanID int
  SupervisionStartDate date
}

Table DepartmentLocations {
  DepartmentID int [pk]
  LocationName varchar [pk]
}

Table Doctors {
  EmployeeID int [pk]
  Specialty varchar
  AcademicDegree varchar
  ScientificArea varchar
  LicenseNumber varchar [unique]
  DepartmentID int
  LocationName varchar
}

Table Nurses {
  EmployeeID int [pk]
  ShiftType varchar
  DepartmentID int
  LocationName varchar
}

Table GeneralStaff {
  EmployeeID int [pk]
  DepartmentID int
  LocationName varchar
  WorkArea varchar
}

Table Rooms {
  RoomNumber int [pk]
  RoomType varchar
  DepartmentID int
  LocationName varchar
}

Table Patients {
  PatientID int [pk]
  NationalID varchar [unique]
  FullName varchar
  DateOfBirth date
  Gender char
  PhoneNumber varchar
  HomeAddress text
  MedicalHistory text
  AdmissionDate date
}

Table PatientInvestigations {
  DoctorID int [pk]
  PatientID int [pk]
  WeeklyConsultationHours int
  InvestigationDate date [pk]
}

Table Appointments {
  AppointmentID int [pk]
  ScheduledDate date
  ScheduledTime time
  AppointmentStatus varchar
  PaymentStatus varchar
  PatientID int
  DoctorID int
}

Table Prescriptions {
  PrescriptionID int [pk]
  MedicationName varchar
  Dosage varchar
  AdministrationInstructions text
  StartDate date
  EndDate date
  PatientID int
  DoctorID int
}

Table MedicalScans {
  ScanID int [pk]
  ScanType varchar
  ScanResult text
  ScanDate date
  PatientID int
  DoctorID int
}

Table VitalSignsLogs {
  LogID int [pk]
  BloodPressure varchar
  HeartRate int
  BodyTemperature decimal
  LogTimestamp datetime
  PatientID int
  NurseID int
}

Table RoomAllocations {
  AllocationID int [pk]
  AdmissionTimestamp datetime
  DischargeTimestamp datetime
  RoomNumber int
  PatientID int
  DoctorID int
}
